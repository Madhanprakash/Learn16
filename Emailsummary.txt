
const items = $input.all();

// Collect summaries from the Aggregate node
const summaries = items.flatMap(item => {
  const data = item.json;

  if (Array.isArray(data.summaries)) {
    return data.summaries;
  }

  if (data.output !== undefined) return [data.output];
  if (data.text !== undefined) return [data.text];

  return [];
});

// Clean summaries
const clean = summaries
  .map(value => String(value ?? "")
    .replace(/```(?:json|text)?/gi, "")
    .replace(/```/g, "")
    .replace(/\s+/g, " ")
    .trim()
  )
  .filter(Boolean);

// Escape HTML so summary content cannot break the layout
const escapeHtml = value => String(value)
  .replace(/&/g, "&amp;")
  .replace(/</g, "&lt;")
  .replace(/>/g, "&gt;")
  .replace(/"/g, "&quot;")
  .replace(/'/g, "&#39;");

// Create numbered summary cards
const cards = clean.map((summary, index) => `
  <tr>
    <td style="padding:12px 14px;border-bottom:1px solid #e8edf3;">
      <table role="presentation" width="100%" cellpadding="0"
        cellspacing="0" border="0">
        <tr>
          <td width="38" valign="top">
            <div style="background:#e8efff;color:#2448a5;
              width:28px;height:28px;line-height:28px;
              text-align:center;border-radius:8px;
              font-weight:bold;font-size:13px;">
              ${index + 1}
            </div>
          </td>
          <td style="font-size:14px;line-height:1.7;
            color:#344054;word-break:break-word;">
            ${escapeHtml(summary)}
          </td>
        </tr>
      </table>
    </td>
  </tr>
`).join("");

const date = new Date().toLocaleDateString("en-IN", {
  day: "numeric",
  month: "short",
  year: "numeric",
  timeZone: "Asia/Kolkata"
});

    subject: `Daily Summary | ${date}`,
    html,
    text,
    count: clean.length
  }
}];
